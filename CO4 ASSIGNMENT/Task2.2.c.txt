#include <stdio.h>
#include <string.h>
 
typedef struct {
    int day, month, year;
} Date;
 
typedef struct {
    char name[40];
    char city[30];
    int pincode;
} Address;
 
typedef struct {
    int id;
    float weight;
    Date booked;
    Address sender;
    Address receiver;
    char service[12];
    float charge;
} Parcel;
 
void printParcel(const Parcel *p)
{
    printf("%d %.1f kg %s-%d %02d/%02d/%04d\\n",
           p->id, p->weight, p->receiver.city,
           p->receiver.pincode, p->booked.day,
           p->booked.month, p->booked.year);
}
 
void sortByWeight(Parcel p[], int n)
{
    int i, j, min;
    Parcel temp;
 
    for (i = 0; i < n - 1; i++) {
        min = i;
        for (j = i + 1; j < n; j++)
            if (p[j].weight < p[min].weight)
                min = j;
 
        if (min != i) {
            temp = p[i];
            p[i] = p[min];
            p[min] = temp;
        }
    }
}
 
void listByCity(const Parcel p[], int n, const char city[])
{
    for (int i = 0; i < n; i++)
        if (strcmp(p[i].receiver.city, city) == 0)
            printParcel(&p[i]);
}
 
int main(void)
{
    Parcel list[3] = {
        {501, 4.5f, {12,10,2026}, {"Asha","Madurai",625001},
         {"Ravi","Madurai",625001}, "normal", 80.0f},
        {502, 1.2f, {12,10,2026}, {"Bala","Chennai",600077},
         {"Kiran","Chennai",600077}, "express", 120.0f},
        {503, 2.8f, {13,10,2026}, {"Devi","Salem",636001},
         {"Meena","Coimbatore",641001}, "normal", 95.0f}
    };
 
    sortByWeight(list, 3);
    for (int i = 0; i < 3; i++)
        printParcel(&list[i]);
 
    printf("\\nParcels for Chennai:\\n");
    listByCity(list, 3, "Chennai");
    return 0;
}
